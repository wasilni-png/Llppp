from telebot import types

def main_menu_keyboard():
    """القائمة الرئيسية"""
    markup = types.ReplyKeyboardMarkup(resize_keyboard=True, row_width=2)
    
    buttons = [
        '🚖 طلب رحلة',
        '📍 إرسال موقعي',
        '💰 رصيدي',
        '📋 رحلاتي',
        '⚙️ إعدادات',
        '📞 الدعم',
        '👤 حسابي',
        '🎫 العروض'
    ]
    
    for i in range(0, len(buttons), 2):
        if i+1 < len(buttons):
            markup.row(buttons[i], buttons[i+1])
        else:
            markup.row(buttons[i])
    
    return markup
